        class AudioController {

            constructor() {

                this.ctx = new (window.AudioContext || window.webkitAudioContext)();

            }

           

            resume() {

                if(this.ctx.state === 'suspended') this.ctx.resume();

            }

            playCorrect() {

                this.resume();

                const osc = this.ctx.createOscillator();

                const gain = this.ctx.createGain();

                osc.connect(gain);

                gain.connect(this.ctx.destination);

               

                osc.type = 'sine';

                osc.frequency.setValueAtTime(523.25, this.ctx.currentTime); // C5

                osc.frequency.exponentialRampToValueAtTime(1046.5, this.ctx.currentTime + 0.1); // C6

               

                gain.gain.setValueAtTime(0.3, this.ctx.currentTime);

                gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.3);

               

                osc.start();

                osc.stop(this.ctx.currentTime + 0.3);

            }

            playWrong() {

                this.resume();

                const osc = this.ctx.createOscillator();

                const gain = this.ctx.createGain();

                osc.connect(gain);

                gain.connect(this.ctx.destination);

               

                osc.type = 'sawtooth';

                osc.frequency.setValueAtTime(150, this.ctx.currentTime);

                osc.frequency.exponentialRampToValueAtTime(80, this.ctx.currentTime + 0.3);

               

                gain.gain.setValueAtTime(0.3, this.ctx.currentTime);

                gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.3);

               

                osc.start();

                osc.stop(this.ctx.currentTime + 0.3);

            }

            playLevelUp() {

                this.resume();

                const notes = [440, 554.37, 659.25, 880]; // A major arpeggio

                notes.forEach((freq, i) => {

                    setTimeout(() => {

                        const osc = this.ctx.createOscillator();

                        const gain = this.ctx.createGain();

                        osc.connect(gain);

                        gain.connect(this.ctx.destination);

                        osc.type = 'triangle';

        class Basketball {

                this.x = x;

                this.y = y;

                this.radius = 45;

                this.value = value;

                this.isGrabbed = false;

                this.offset = Math.random() * Math.PI * 2; // For sine wave floating

               

                // Speed based on level

                this.speed = config.baseSpeed + (state.level * 0.2) + (Math.random() * 0.5);

                this.rotation = Math.random() * Math.PI;

                this.rotSpeed = (Math.random() - 0.5) * 0.02;

            }

            update(dt) {

                if (!this.isGrabbed) {

                    this.y -= this.speed * (dt / 16);

                    this.x += Math.sin(this.y * 0.02 + this.offset) * 1.5;

                    this.rotation += this.rotSpeed;

                   

                    // Recycle if floating past top

                    if (this.y < -100) {

                        this.y = config.height + 100 + Math.random() * 200;

                        this.x = 100 + Math.random() * (config.width - 200);

                }

            }

        const QuestionManager = {

            generate(level) {

                let a, b, c, x, op;

                const types = [];

                if (level >= 1) types.push('+');

                if (level >= 2) types.push('-');

                if (level >= 3) types.push('*');

                if (level >= 4) types.push('/');

                if (level >= 5) types.push('all');

               

                op = types[Math.floor(Math.random() * types.length)];

                if (op === 'all') {

                    const allOps = ['+', '-', '*', '/'];

                    op = allOps[Math.floor(Math.random() * allOps.length)];

                }

               

                const maxVal = 10 + (level * 5);

                // ฟังก์ชันสุ่มตัวเลขที่สามารถให้ค่าติดลบได้

                function getRand(max, allowNegative = true, avoidZero = false) {

                    let min = allowNegative ? -max : 1;

                    let val = Math.floor(Math.random() * (max - min + 1)) + min;

                    if (avoidZero && val === 0) val = 1;

                    return val;

                }

                // ระดับสูงขึ้นอนุญาตให้มีเลขติดลบ

                const allowNeg = level >= 1;

               

                switch(op) {

                    case '+':

                        a = getRand(maxVal, allowNeg);

                        x = getRand(maxVal, allowNeg);

                        c = a + x;

                        break;

                    case '-':

                        a = getRand(maxVal, allowNeg);

                        x = getRand(maxVal, allowNeg);

                        c = a - x;

                        break;

                    case '*':

                        a = getRand(Math.min(maxVal, 12), allowNeg, true);

                        x = getRand(Math.min(maxVal, 12), allowNeg);

                        c = a * x;

                        break;

                    case '/':

                        a = getRand(Math.min(maxVal, 12), allowNeg, true);

                        c = getRand(Math.min(maxVal, 12), allowNeg);

                        x = a * c;

                        break;

                }

               

                let equationStr = "";

                // จัดรูปแบบสมการ หากตัวเลขติดลบให้ใส่วงเล็บ เพื่อความสวยงามทางคณิตศาสตร์

                const formatNum = (num) => num < 0 ? `(${num})` : num;

                if (Math.random() > 0.5 && op !== '-') {

                    equationStr = `x ${op} ${formatNum(a)} = ${c}`;

                } else {

                    equationStr = `${formatNum(a)} ${op} x = ${c}`;

                }

               

                let answers = [x];

                while(answers.length < 4 + Math.floor(level/2)) {

                    // Generate plausible wrong answers (สามารถติดลบได้)

                    let variation = getRand(10, true, true);

                    let wrong = x + variation;

                    if (!answers.includes(wrong)) {

                        answers.push(wrong);

                    }

                }

               

                // Shuffle choices

                answers.sort(() => Math.random() - 0.5);

               

                return { str: equationStr, answer: x, choices: answers };

            }

        };

        function updateCursorFromEvent(e) {

            if (state.isARMode) return;

            const rect = canvas.getBoundingClientRect();

            const scaleX = canvas.width / rect.width;

            const scaleY = canvas.height / rect.height;

           

            // Handle touch or mouse

            let clientX = e.clientX;

            let clientY = e.clientY;

            if(e.touches && e.touches.length > 0) {

                clientX = e.touches[0].clientX;

                clientY = e.touches[0].clientY;

            }

            cursor.x = (clientX - rect.left) * scaleX;

            cursor.y = (clientY - rect.top) * scaleY;

        }

        canvas.addEventListener('mousedown', (e) => { cursor.isDown = true; updateCursorFromEvent(e); });

        canvas.addEventListener('mousemove', (e) => { if(cursor.isDown) updateCursorFromEvent(e); else updateCursorFromEvent(e); });

        canvas.addEventListener('mouseup', () => { cursor.isDown = false; });

        canvas.addEventListener('mouseleave', () => { cursor.isDown = false; });

        canvas.addEventListener('touchstart', (e) => { e.preventDefault(); cursor.isDown = true; updateCursorFromEvent(e); }, {passive: false});

        canvas.addEventListener('touchmove', (e) => { e.preventDefault(); updateCursorFromEvent(e); }, {passive: false});

        canvas.addEventListener('touchend', (e) => { e.preventDefault(); cursor.isDown = false; }, {passive: false});

        function onARResults(results) {

            if (!state.isARMode || state.status !== 'playing') return;

            if (results.multiHandLandmarks && results.multiHandLandmarks.length > 0) {

                const landmarks = results.multiHandLandmarks[0];

                const thumbTip = landmarks[4];

                const indexTip = landmarks[8];

               

                // Front camera is usually mirrored. X coordinate mapping:

                // We will mirror the video when drawing, so we must mirror coordinates here so visual matches logic.

                let x = (1 - indexTip.x) * config.width;

                let y = indexTip.y * config.height;

               

                // Pinch detection (2D Euclidean distance)

                const dist = Math.hypot(thumbTip.x - indexTip.x, thumbTip.y - indexTip.y);

                const isPinching = dist < 0.08; // Threshold for pinch

               

                // Smooth cursor movement a bit

                cursor.x += (x - cursor.x) * 0.5;

                cursor.y += (y - cursor.y) * 0.5;

                cursor.isDown = isPinching;

            } else {

                cursor.isDown = false;

            }

        }

        function updateLogic(dt) {

            if (state.status !== 'playing') return;

            // 1. Cursor Interaction Logic

            if (cursor.isDown && !cursor.wasDown) {

                // Just pressed/pinched - try to grab

                for (let i = balls.length - 1; i >= 0; i--) {

                    const b = balls[i];

                    const hitDist = Math.hypot(b.x - cursor.x, b.y - cursor.y);

                    if (hitDist < b.radius + 30) { // generous hitbox

                        grabbedBall = b;

                        b.isGrabbed = true;

                        break;

                    }

                }

            }

            if (cursor.isDown && grabbedBall) {

                // Dragging

                grabbedBall.x = cursor.x;

                grabbedBall.y = cursor.y;

            }

            if (!cursor.isDown && cursor.wasDown) {

                // Just released/unpinched

                if (grabbedBall) {

                    // Check if dropped near hoop center

                    const dropDist = Math.hypot(grabbedBall.x - hoop.x, grabbedBall.y - hoop.y);

                   

                    if (dropDist < hoop.radius + 40) {

                        // Dropped in hoop!

                        let val = grabbedBall.value;

                        let bx = grabbedBall.x;

                        let by = grabbedBall.y;

                       

                        // Remove this ball from array

                        balls = balls.filter(b => b !== grabbedBall);

                       

                        checkAnswer(val, bx, by);

                    }

                   

                    if (grabbedBall) grabbedBall.isGrabbed = false;

                    grabbedBall = null;

                }

            }

            cursor.wasDown = cursor.isDown;

            // 2. Update Entities

            balls.forEach(b => b.update(dt));

            particles.forEach(p => p.update());

            floatingTexts.forEach(ft => ft.update());

            // Cleanup dead entities

            particles = particles.filter(p => p.life > 0);

            floatingTexts = floatingTexts.filter(ft => ft.life > 0);

        }

        function drawCourt(ctx) {

            // Hardwood floor

            ctx.fillStyle = '#b47b44';

            ctx.fillRect(0, 0, config.width, config.height);

           

            // Planks

            ctx.strokeStyle = '#a36a33';

            ctx.lineWidth = 2;

            for(let i=0; i<config.height; i+=40) {

                ctx.beginPath(); ctx.moveTo(0, i); ctx.lineTo(config.width, i); ctx.stroke();

            }

           

            // Paint Area

            ctx.fillStyle = '#8a2323';

            ctx.fillRect(config.width/2 - 200, 0, 400, 300);

           

            // Lines

            ctx.strokeStyle = '#ffffff';

            ctx.lineWidth = 6;

            ctx.strokeRect(config.width/2 - 200, 0, 400, 300); // Paint box

           

            // Free throw half circle

            ctx.beginPath();

            ctx.arc(config.width/2, 300, 150, 0, Math.PI);

            ctx.stroke();

        }

        function drawHoop(ctx) {

            // Glow effect if hovering with grabbed ball

            let isHovering = false;

            if (grabbedBall) {

                const dropDist = Math.hypot(grabbedBall.x - hoop.x, grabbedBall.y - hoop.y);

                if (dropDist < hoop.radius + 40) isHovering = true;

            }

            // Backboard

            ctx.fillStyle = 'rgba(255, 255, 255, 0.9)';

            ctx.fillRect(hoop.x - hoop.width/2, 20, hoop.width, hoop.height);

            ctx.strokeStyle = '#dc2626'; // red-600

            ctx.lineWidth = 8;

            ctx.strokeRect(hoop.x - hoop.width/2, 20, hoop.width, hoop.height);

           

            // Inner box

            ctx.strokeRect(hoop.x - 50, 70, 100, 80);

           

            // Net (Draw first so it's behind rim front)

            ctx.strokeStyle = '#ffffff';

            ctx.lineWidth = 3;

            ctx.beginPath();

            for(let i=-60; i<=60; i+=20) {

                ctx.moveTo(hoop.x + i, hoop.y);

                ctx.lineTo(hoop.x + i * 0.5, hoop.y + 90);

            }

            for(let i=20; i<=90; i+=20) {

                ctx.moveTo(hoop.x - 60 + (i*0.3), hoop.y + i);

                ctx.lineTo(hoop.x + 60 - (i*0.3), hoop.y + i);

            }

            ctx.stroke();

            // Rim

            ctx.beginPath();

            ctx.ellipse(hoop.x, hoop.y, hoop.radius, hoop.radius * 0.3, 0, 0, Math.PI * 2);

            ctx.strokeStyle = isHovering ? '#fde047' : '#ea580c'; // yellow if hover, orange normal

            if (isHovering) {

                ctx.shadowColor = '#fde047';

                ctx.shadowBlur = 20;

            }

            ctx.lineWidth = 12;

            ctx.stroke();

            ctx.shadowBlur = 0; // reset

        }

        function drawCursor(ctx) {

            if (cursor.x < 0) return; // offscreen

           

            ctx.save();

            ctx.translate(cursor.x, cursor.y);

           

            if (state.isARMode) {

                // AR Hand Cursor (Pinch indicator)

                ctx.beginPath();

                ctx.arc(0, 0, cursor.isDown ? 15 : 25, 0, Math.PI*2);

                ctx.fillStyle = cursor.isDown ? 'rgba(52, 211, 153, 0.6)' : 'rgba(255, 255, 255, 0.4)';

                ctx.fill();

                ctx.lineWidth = 4;

                ctx.strokeStyle = cursor.isDown ? '#10b981' : '#ffffff';

                ctx.stroke();

               

                // Crosshairs

                ctx.beginPath();

                ctx.moveTo(-35, 0); ctx.lineTo(35, 0);

                ctx.moveTo(0, -35); ctx.lineTo(0, 35);

                ctx.lineWidth = 2;

                ctx.stroke();

            } else {

                // Mouse/Touch simple highlight when grabbing

                if (cursor.isDown && !grabbedBall) {

                    ctx.beginPath();

                    ctx.arc(0, 0, 20, 0, Math.PI*2);

                    ctx.fillStyle = 'rgba(255, 255, 255, 0.2)';

                    ctx.fill();

                }

            }

           

            ctx.restore();

        }

        function draw() {

            // Clear canvas

            ctx.clearRect(0, 0, config.width, config.height);

            if (state.isARMode && state.cameraActive) {

                // Draw mirrored webcam feed as background

                ctx.save();

                ctx.scale(-1, 1);

                // The video object-fit is cover/fill equivalent mapping to canvas

                ctx.drawImage(videoElement, -config.width, 0, config.width, config.height);

                ctx.restore();

               

                // Darken slightly for readability

                ctx.fillStyle = 'rgba(0, 0, 0, 0.3)';

                ctx.fillRect(0, 0, config.width, config.height);

            } else {

                drawCourt(ctx);

            }

            drawHoop(ctx);

            // Draw Balls

            balls.forEach(b => {

                if (!b.isGrabbed) b.draw(ctx);

            });

            // Draw grabbed ball last (on top)

            if (grabbedBall) grabbedBall.draw(ctx);

            // Draw Effects

            particles.forEach(p => p.draw(ctx));

            floatingTexts.forEach(ft => ft.draw(ctx));

            // Draw Custom Cursor

            drawCursor(ctx);

           

            // Paused Overlay

            if(state.status === 'paused') {

                ctx.fillStyle = 'rgba(0,0,0,0.6)';

                ctx.fillRect(0,0,config.width,config.height);

                ctx.fillStyle = 'white';

                ctx.font = 'bold 60px Kanit';

                ctx.textAlign = 'center';

                ctx.fillText('PAUSED', config.width/2, config.height/2);

            }

        }

        function gameLoop(timestamp) {

            if (!lastTime) lastTime = timestamp;

            const dt = timestamp - lastTime;

            lastTime = timestamp;

            updateLogic(dt);

            draw();

            animationId = requestAnimationFrame(gameLoop);

        }

        function updateUI() {

            uiScore.innerText = state.score;

            uiLevel.innerText = `Level ${state.level}`;

           

            // Time format MM:SS

            let m = Math.floor(state.time / 60).toString().padStart(2, '0');

            let s = (state.time % 60).toString().padStart(2, '0');

            uiTime.innerText = `${m}:${s}`;

           

            // Combo

            if(state.combo >= 3) {

                uiCombo.innerText = `${state.combo} COMBO 🔥`;

                uiCombo.classList.add('pulse-glow', 'text-pink-400');

            } else {

                uiCombo.innerText = '';

                uiCombo.classList.remove('pulse-glow');

            }

           

            // Urgency color for time

            if (state.time <= 30) {

                uiTime.classList.remove('text-emerald-400');

                uiTime.classList.add('text-red-500', 'animate-pulse');

            } else {

                uiTime.classList.add('text-emerald-400');

                uiTime.classList.remove('text-red-500', 'animate-pulse');

            }

        }

        async function startGame(useAR) {

            audio.resume(); // Ensure audio context is ready on user interaction

            state.isARMode = useAR;

           

            const nameInput = document.getElementById('playerNameInput');

            state.playerName = nameInput.value.trim() || 'ผู้เล่นนิรนาม';

            if (useAR) {

                let success = await initAR();

                if (!success) {

                    state.isARMode = false; // Fallback

                }

            }

            document.getElementById('startOverlay').classList.add('hidden');

            document.getElementById('btnPause').classList.remove('hidden');

            document.getElementById('btnEnd').classList.remove('hidden');

           

            state.status = 'playing';

            state.score = 0;

            state.level = 1;

            state.time = config.initialTime;

            state.combo = 0;

            state.stats = { correct: 0, wrong: 0 };

           

            spawnQuestion();

            updateUI();

           

            // Start timer

            clearInterval(timerInterval);

            timerInterval = setInterval(() => {

                if (state.status === 'playing') {

                    state.time--;

                    updateUI();

                    if (state.time <= 0) {

                        endGame();

                    }

                }

            }, 1000);

            if (!animationId) {

                lastTime = performance.now();

                animationId = requestAnimationFrame(gameLoop);

            }

        }

        function togglePause() {

            if(state.status === 'playing') {

                state.status = 'paused';

                document.getElementById('btnPause').innerText = '▶ เล่นต่อ (Resume)';

            } else if (state.status === 'paused') {

                state.status = 'playing';

                document.getElementById('btnPause').innerText = '⏸ หยุดเกม (Pause)';

                lastTime = performance.now(); // reset delta time

            }

        }

        function endGame() {

            state.status = 'gameover';

            clearInterval(timerInterval);

           

            // Stop Camera if active

            if(camera && state.cameraActive) {

                camera.stop();

                state.cameraActive = false;

            }

            Leaderboard.save(state.playerName, state.score);

           

            const lbList = document.getElementById('leaderboardList');

            lbList.innerHTML = '';

            const topScores = Leaderboard.get();

            if (topScores.length === 0) {

                lbList.innerHTML = '<li class="text-center text-gray-400">ยังไม่มีข้อมูลคะแนน</li>';

            } else {

                topScores.forEach((entry, i) => {

                    let medal = i === 0 ? '🥇' : i === 1 ? '🥈' : i === 2 ? '🥉' : '🔹';

                    let bg = i === 0 ? 'bg-amber-500/20 border border-amber-500/50' : 'bg-white/10';

                    lbList.innerHTML += `

                        <li class="flex justify-between items-center ${bg} p-2 px-3 rounded-lg">

                            <span class="flex items-center gap-2"><span class="text-lg">${medal}</span> <span class="font-semibold text-white truncate max-w-[150px]">${entry.name}</span></span>

                            <span class="font-black text-amber-400">${entry.score} pts</span>

                        </li>`;

                });

            }

           

            // Populate Game Over Modal

            document.getElementById('finalScore').innerText = state.score;

            document.getElementById('finalCorrect').innerText = state.stats.correct;

            document.getElementById('finalWrong').innerText = state.stats.wrong;

           

            let total = state.stats.correct + state.stats.wrong;

            let acc = total > 0 ? Math.round((state.stats.correct / total) * 100) : 0;

            document.getElementById('finalAccuracy').innerText = `${acc}%`;

           

            // Badges System

            const badgeContainer = document.getElementById('badgeContainer');

            const badgeIcon = document.getElementById('badgeIcon');

            const badgeName = document.getElementById('badgeName');

           

            badgeContainer.classList.remove('hidden');

            if (state.score >= 500 && acc >= 80) {

                badgeIcon.innerText = '👑';

                badgeName.innerText = 'Math Master (ปรมาจารย์)';

            } else if (state.score >= 250) {

                badgeIcon.innerText = '⭐';

                badgeName.innerText = 'Math Star (ดาวรุ่ง)';

            } else if (state.score > 0) {

                badgeIcon.innerText = '🏅';

                badgeName.innerText = 'Math Rookie (มือใหม่)';

            } else {

                badgeContainer.classList.add('hidden');

            }

            document.getElementById('gameOverModal').classList.remove('hidden');

        }

        function restartGame() {

            document.getElementById('gameOverModal').classList.add('hidden');

            startGame(state.isARMode); // Restart in same mode

        }
